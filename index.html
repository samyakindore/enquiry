function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Target 'Sheet1' (or change to 'Sheet2' if you are sending data there) var sheet = ss.getSheetByName("Sheet1"); if (!sheet) { return ContentService .createTextOutput(JSON.stringify({ "result": "error", "error": "Sheet tab not found!" })) .setMimeType(ContentService.MimeType.JSON); } // Parse parameters sent via URLSearchParams / POST body var p = e.parameter || {}; // 10 exact headers matching your Google Sheet layout var headers = [ "Enquiry ID", "Date", "Student Name", "Mobile No", "Email Id", "Qualification", "Course", "Purpose", "Preferred Timing", "Source" ]; // Add headers if the sheet is completely blank if (sheet.getLastRow() === 0) { sheet.appendRow(headers); } // Generate Auto Enquiry ID and Date var tz = ss.getSpreadsheetTimeZone() || "Asia/Kolkata"; var enquiryId = "ENQ-" + (1000 + sheet.getLastRow()); var formattedDate = Utilities.formatDate(new Date(), tz, "dd-MMM-yyyy"); // Map the incoming HTML form field names (`name="..."`) directly var newRow = [ enquiryId, // Column A: Enquiry ID formattedDate, // Column B: Date p.studentname || "", // Column C: Student Name p.mobileno || "", // Column D: Mobile No p.emailid || "", // Column E: Email Id p.qualification || "", // Column F: Qualification p.coreinterest || "", // Column G: Course p.purpose || "", // Column H: Purpose p.preferredtiming || "", // Column I: Preferred Timing p.leadsource || "" // Column J: Source ]; // Append entry to sheet sheet.appendRow(newRow); return ContentService .createTextOutput(JSON.stringify({ "result": "success", "enquiryId": enquiryId })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ "result": "error", "error": err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
